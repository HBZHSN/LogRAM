cmake_minimum_required(VERSION 3.20)
project(LogRAMNpp LANGUAGES CXX RC)
add_executable(LogRAMNppRouterTest LogOpenRouterTest.cpp LogOpenRouter.cpp)
target_compile_features(LogRAMNppRouterTest PRIVATE cxx_std_17)
target_compile_definitions(LogRAMNppRouterTest PRIVATE UNICODE _UNICODE)
add_library(LogRAMNpp SHARED Plugin.cpp LogOpenRouter.cpp Version.rc LogRAM.Npp.def)
set_target_properties(LogRAMNpp PROPERTIES OUTPUT_NAME "LogRAM.Npp" PREFIX "")
target_compile_features(LogRAMNpp PRIVATE cxx_std_17)
target_compile_definitions(LogRAMNpp PRIVATE UNICODE _UNICODE)
target_link_libraries(LogRAMNpp PRIVATE user32 comctl32)
if(MSVC)
    set_property(TARGET LogRAMNppRouterTest PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    set_property(TARGET LogRAMNpp PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(LogRAMNpp PRIVATE /W4 /utf-8)
else()
    target_link_options(LogRAMNppRouterTest PRIVATE -static -static-libgcc -static-libstdc++)
    target_link_options(LogRAMNpp PRIVATE -static -static-libgcc -static-libstdc++)
    target_compile_options(LogRAMNpp PRIVATE -Wall -Wextra)
endif()
